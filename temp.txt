Suppose in a row user gives internal IP as 9000 and reverse proxy name as route0 and in another row internal ip 9001 and proxy name route1 then

then nginx configuration should be

server {
    listen 80;
    server_name calcutta-hacks-server.uemcseaiml.org;

    # Proxy requests to 127.0.0.1:9000
    location /route0/ {
        rewrite ^/route0(/.*)$ $1 break;
        proxy_pass http://127.0.0.1:9000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # Proxy requests to 127.0.0.1:9001
    location /route1/ {
        rewrite ^/route1(/.*)$ $1 break;
        proxy_pass http://127.0.0.1:9001;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}

so the location block is repeated and nested inside the server block for each of the rows user gives in the frontend
